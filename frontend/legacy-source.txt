async function summarizePdf(file){const pdf=await window.pdfjsLib.getDocument({data:await file.arrayBuffer()}).promise;const text=(await extractText(pdf)).join(' ');const words=textToWords(text);const sentences=text.split(/(?<=[.!?])\s+/).filter(Boolean);const freq={};for(const w of words.map(x=>x.toLowerCase().replace(/[^a-z0-9]/g,''))){if(w.length>3)freq[w]=(freq[w]||0)+1}const top=new Set(Object.entries(freq).sort((a,b)=>b[1]-a[1]).slice(0,15).map(([w])=>w));const scored=sentences.map(s=>[s,[...top].filter(w=>s.toLowerCase().includes(w)).length]).sort((a,b)=>b[1]-a[1]).slice(0,8).map(x=>x[0]);const summary=scored.join(' ')||text.slice(0,3000);showModal('Summarize PDF',`<div class="status show">Local extractive summary — no document is uploaded to an AI provider.</div><div class="result-box">${escapeHtml(summary)}</div>`,'Download Summary',()=>downloadText(summary,'KynetraPDF_Summary.txt'))}

async function notesPdf(file){const pdf=await window.pdfjsLib.getDocument({data:await file.arrayBuffer()}).promise;const text=(await extractText(pdf)).join('\n');const words=textToWords(text);const freq={};for(const w of words.map(x=>x.toLowerCase().replace(/[^a-z0-9]/g,''))){if(w.length>4)freq[w]=(freq[w]||0)+1}const terms=Object.entries(freq).sort((a,b)=>b[1]-a[1]).slice(0,20).map(x=>x[0]);const notes=`KynetraPDF Study/Research Notes\n\nKEY TERMS\n- ${terms.join('\n- ')}\n\nDOCUMENT SUMMARY\n${text.slice(0,6000)}\n\nREVISION PROMPTS\n1. What is the central topic of this document?\n2. What methods or evidence are presented?\n3. What are the main results or conclusions?\n4. What limitations or research gaps are mentioned?`;showModal('Generate Notes',`<div class="result-box">${escapeHtml(notes)}</div>`,'Download Notes',()=>downloadText(notes,'KynetraPDF_Notes.txt'))}

async function chatPdf(file){const pdf=await window.pdfjsLib.getDocument({data:await file.arrayBuffer()}).promise;const text=(await extractText(pdf)).join('\n');showModal('Chat with PDF',`<div class="field"><label>Question</label><textarea id="chatQ" placeholder="e.g. What is the research objective?"></textarea></div><div class="status show">Browser-local answer mode searches the extracted text. Add an AI endpoint for generative answers.</div><div id="chatAnswer" class="result-box">Ask a question.</div>`,'Ask',()=>{const q=$('#chatQ').value.trim();const ans=localQuestionAnswer(q,text);$('#chatAnswer').textContent=ans});}
function localQuestionAnswer(q,text){if(!q)return 'Please enter a question.';const terms=q.toLowerCase().split(/\W+/).filter(x=>x.length>3);const chunks=text.split(/\n|(?<=[.!?])\s+/).filter(Boolean);const scored=chunks.map(s=>[s,terms.reduce((n,t)=>n+(s.toLowerCase().includes(t)?1:0),0)]).sort((a,b)=>b[1]-a[1]).slice(0,5).filter(x=>x[1]>0);return scored.length?scored.map(x=>x[0]).join('\n\n'):'No closely matching text was found. For semantic/generative chat, connect an AI endpoint.'}

async function translatePdf(file){const pdf=await window.pdfjsLib.getDocument({data:await file.arrayBuffer()}).promise;const text=(await extractText(pdf)).join('\n');showModal('Translate PDF',`<div class="form-row"><div class="field"><label>Target language</label><input id="lang" value="Hindi" placeholder="e.g. Bengali, Hindi, English"></div><div class="field"><label>Translation endpoint (optional)</label><input id="transEndpoint" placeholder="https://your-server/translate"></div></div><div class="status show">A browser cannot reliably call arbitrary translation services without an endpoint because of API/CORS constraints. Provide your own endpoint to enable live translation.</div>`,'Translate',async()=>{const ep=$('#transEndpoint').value.trim();const lang=$('#lang').value.trim();if(!ep){downloadText(text,'KynetraPDF_Original_Text.txt');showToast('No translation endpoint configured; exported extracted text instead.');return}try{const r=await fetch(ep,{method:'POST',headers:{'Content-Type':'application/json'},body:JSON.stringify({text,source:'auto',target:lang})});if(!r.ok)throw Error('Translation endpoint returned '+r.status);const j=await r.json();const translated=j.translatedText||j.text||JSON.stringify(j);downloadText(translated,'KynetraPDF_Translated.txt');showToast('Translated text downloaded.')}catch(e){showToast('Translation failed: '+e.message)}})}
